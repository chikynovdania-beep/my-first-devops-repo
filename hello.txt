1. Посмотреть ветки git branch          # список локальных веток (* = текущая)
git branch -v       # с хешем и последним коммитом
git branch -a       # все ветки (локальные + удалённые) 2. Создать веткуgit branch имя-ветки        # создаёт, но не переключается
git switch -c имя-ветки     # создаёт И переключается (рекомендуется)
# или старый способ:
git checkout -b имя-ветки 3. Переключиться между ветками git switch master           # переключиться на master
git switch feature-login    # переключиться на существующую ветку 
 4. Слить ветку (merge) git switch master           # перейти в целевую ветку
git merge feature-login     # слить feature-login в master 5. Удалить ветку git branch -d feature-login     # безопасное удаление (если слита)
git branch -D feature-login     # принудительное удаление (осторожно!) 7. Посмотреть, какие ветки уже слиты / не слиты git branch --merged       # ветки, полностью включённые в текущую
git branch --no-merged    # ветки с уникальными коммитами Отлично подходит для чистки: git branch --merged | grep -v master | xargs git branch -d (удалит все слитые ветки, кроме master)
